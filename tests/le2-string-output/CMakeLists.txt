cmake_minimum_required(VERSION 3.20)
project(mevr_le2_string_output_tests LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Pinned FString and strict Windows conversion checks require Win64")
endif()
enable_testing()
set(source ../../src/core/qa/BoundedUtf16.cpp)
add_library(le2-string-output-production OBJECT ${source})
add_executable(le2-string-output-tests StringTests.cpp ${source})
foreach(target le2-string-output-production le2-string-output-tests)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE ../../src)
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
endforeach()
target_compile_definitions(le2-string-output-tests PRIVATE MEVR_BOUNDED_UTF16_TEST)
add_test(NAME le2-string-output COMMAND le2-string-output-tests)
set_tests_properties(le2-string-output PROPERTIES TIMEOUT 30)

set(MEVR_STRING_SDK_DIR "" CACHE PATH "Optional pinned LExSDKv2 checkout for actual header regression")
if(MEVR_STRING_SDK_DIR)
    add_executable(le2-string-output-sdk-tests SdkHeaderTests.cpp ${source})
    target_compile_features(le2-string-output-sdk-tests PRIVATE cxx_std_20)
    target_include_directories(le2-string-output-sdk-tests PRIVATE ../../src)
    target_include_directories(le2-string-output-sdk-tests SYSTEM PRIVATE "${MEVR_STRING_SDK_DIR}/Src" "${MEVR_STRING_SDK_DIR}/External")
    target_compile_definitions(le2-string-output-sdk-tests PRIVATE SDK_TARGET_LE2 NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_precompile_headers(le2-string-output-sdk-tests PRIVATE "${MEVR_STRING_SDK_DIR}/Src/LESDK/_Global.pch.hpp")
    target_compile_options(le2-string-output-sdk-tests PRIVATE /W4 /WX /EHsc /bigobj /permissive /wd4121 /wd26495 /wd4702)
    add_test(NAME le2-string-output-sdk COMMAND le2-string-output-sdk-tests)
    set_tests_properties(le2-string-output-sdk PROPERTIES TIMEOUT 30)
endif()
