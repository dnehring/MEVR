cmake_minimum_required(VERSION 3.20)
project(mevr_le2_console_input_tests LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Console input tests exercise the actual Windows conversion API")
endif()
enable_testing()
add_executable(le2-console-input-tests InputTests.cpp)
target_compile_features(le2-console-input-tests PRIVATE cxx_std_20)
target_compile_definitions(le2-console-input-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(le2-console-input-tests PRIVATE ../../src)
if(MSVC)
    target_compile_options(le2-console-input-tests PRIVATE /W4 /WX)
else()
    target_compile_options(le2-console-input-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME le2-console-input COMMAND le2-console-input-tests)
