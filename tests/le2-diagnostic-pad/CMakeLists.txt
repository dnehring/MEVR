cmake_minimum_required(VERSION 3.20)
project(mevr_le2_diagnostic_pad_tests LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(le2-diagnostic-pad-tests PadTests.cpp ../../src/core/qa/DiagnosticPad.cpp)
target_compile_features(le2-diagnostic-pad-tests PRIVATE cxx_std_20)
target_include_directories(le2-diagnostic-pad-tests PRIVATE ../../src)
target_link_libraries(le2-diagnostic-pad-tests PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(le2-diagnostic-pad-tests PRIVATE /W4 /WX /EHsc)
else()
    target_compile_options(le2-diagnostic-pad-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME le2-diagnostic-pad COMMAND le2-diagnostic-pad-tests)
