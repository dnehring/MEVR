cmake_minimum_required(VERSION 3.20)
project(mevr_le2_diagnostic_bridge_tests LANGUAGES CXX)
if(NOT MSVC)
    message(FATAL_ERROR "Actual Windows XInput bridge requires MSVC")
endif()
add_executable(le2-diagnostic-bridge-tests BridgeTests.cpp ../../src/game/GamepadBridge.cpp ../../src/core/qa/DiagnosticPad.cpp)
target_compile_features(le2-diagnostic-bridge-tests PRIVATE cxx_std_20)
target_include_directories(le2-diagnostic-bridge-tests PRIVATE ../../src)
target_compile_definitions(le2-diagnostic-bridge-tests PRIVATE MEVR_GAME_LE2 MEVR_LE2_DIAGNOSTIC MEVR_DIAGNOSTIC_PAD_BRIDGE_TEST NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(le2-diagnostic-bridge-tests PRIVATE /W4 /WX /EHsc)
# Compile all real Win32 lookup/provenance code without installing any hook.
add_library(le2-diagnostic-bridge-production OBJECT ../../src/game/GamepadBridge.cpp)
target_compile_features(le2-diagnostic-bridge-production PRIVATE cxx_std_20)
target_include_directories(le2-diagnostic-bridge-production PRIVATE ../../src)
target_compile_definitions(le2-diagnostic-bridge-production PRIVATE MEVR_GAME_LE2 MEVR_LE2_DIAGNOSTIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(le2-diagnostic-bridge-production PRIVATE /W4 /WX /EHsc)
set(MEVR_BRIDGE_OPENXR_INCLUDE_DIR "" CACHE PATH "Optional initialized OpenXR include directory for ordinary trilogy compile checks")
if(MEVR_BRIDGE_OPENXR_INCLUDE_DIR)
    foreach(game LE1 LE2 LE3)
        add_library(le2-bridge-ordinary-${game} OBJECT ../../src/game/GamepadBridge.cpp)
        target_compile_features(le2-bridge-ordinary-${game} PRIVATE cxx_std_20)
        target_include_directories(le2-bridge-ordinary-${game} PRIVATE ../../src "${MEVR_BRIDGE_OPENXR_INCLUDE_DIR}")
        target_compile_definitions(le2-bridge-ordinary-${game} PRIVATE MEVR_GAME_${game} NOMINMAX WIN32_LEAN_AND_MEAN)
        target_compile_options(le2-bridge-ordinary-${game} PRIVATE /W4 /WX /EHsc)
    endforeach()
endif()
enable_testing()
add_test(NAME le2-diagnostic-bridge COMMAND le2-diagnostic-bridge-tests)
