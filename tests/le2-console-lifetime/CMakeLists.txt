cmake_minimum_required(VERSION 3.20)
project(mevr_le2_console_lifetime_tests LANGUAGES CXX)
enable_testing()
add_executable(le2-console-lifetime-tests LifetimeTests.cpp)
target_compile_features(le2-console-lifetime-tests PRIVATE cxx_std_20)
target_include_directories(le2-console-lifetime-tests PRIVATE ../../src)
if(MSVC)
    target_compile_options(le2-console-lifetime-tests PRIVATE /W4 /WX)
else()
    target_compile_options(le2-console-lifetime-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME le2-console-lifetime COMMAND le2-console-lifetime-tests)

# Optional actual-SDK, empty-string ABI checks. They use only declarations and
# no-allocation stubs, never link or call the engine allocator.
set(MEVR_LIFETIME_SDK_DIR "" CACHE PATH "Pinned LExSDKv2 checkout for MSVC x64 packed ABI checks")
if(MEVR_LIFETIME_SDK_DIR)
    if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "SDK packed ABI checks require MSVC x64")
    endif()
    foreach(game IN ITEMS LE2 LE3)
        set(target le2-console-lifetime-sdk-${game})
        add_executable(${target} SdkLifetimeTests.cpp)
        target_compile_features(${target} PRIVATE cxx_std_20)
        target_compile_definitions(${target} PRIVATE SDK_TARGET_${game} NOMINMAX _CRT_SECURE_NO_WARNINGS)
        target_include_directories(${target} PRIVATE ../../src)
        target_include_directories(${target} SYSTEM PRIVATE "${MEVR_LIFETIME_SDK_DIR}/Src" "${MEVR_LIFETIME_SDK_DIR}/External")
        target_precompile_headers(${target} PRIVATE "${MEVR_LIFETIME_SDK_DIR}/Src/LESDK/_Global.pch.hpp")
        target_compile_options(${target} PRIVATE /W4 /WX /bigobj /permissive /wd4121 /wd26495 /wd4702)
        add_test(NAME ${target} COMMAND ${target})
    endforeach()
endif()
