cmake_minimum_required(VERSION 3.20)
project(mevr_le2_process_event_diagnostic_tests LANGUAGES CXX)
enable_testing()
add_executable(le2-process-event-diagnostic-tests ChainTests.cpp)
target_compile_features(le2-process-event-diagnostic-tests PRIVATE cxx_std_20)
target_include_directories(le2-process-event-diagnostic-tests PRIVATE ../../src)
if(MSVC)
    target_compile_options(le2-process-event-diagnostic-tests PRIVATE /W4 /WX)
    # Compile the real production adapter, without linking or installing hooks.
    add_library(le2-process-event-diagnostic-compile OBJECT ../../src/plugin/ProcessEventHook.cpp)
    target_compile_features(le2-process-event-diagnostic-compile PRIVATE cxx_std_20)
    target_include_directories(le2-process-event-diagnostic-compile PRIVATE ../../src)
    target_compile_definitions(le2-process-event-diagnostic-compile PRIVATE MEVR_GAME_LE2 MEVR_LE2_DIAGNOSTIC NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(le2-process-event-diagnostic-compile PRIVATE /W4 /WX /EHsc)
else()
    target_compile_options(le2-process-event-diagnostic-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME le2-process-event-diagnostic COMMAND le2-process-event-diagnostic-tests)
add_executable(le2-process-event-witness-tests WitnessTests.cpp)
target_compile_features(le2-process-event-witness-tests PRIVATE cxx_std_20)
target_include_directories(le2-process-event-witness-tests PRIVATE ../../src)
if(MSVC)
    target_compile_options(le2-process-event-witness-tests PRIVATE /W4 /WX)
else()
    target_compile_options(le2-process-event-witness-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME le2-process-event-witness COMMAND le2-process-event-witness-tests)
